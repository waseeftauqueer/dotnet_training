Display the list of customers who resides in Bangalore

SELECT * FROM customers
WHERE caddress = 'Bangalore';

Display the list of customers who does not reside in Bangalore or Chennai.

SELECT * FROM customers
WHERE caddress NOT IN ('Bangalore', 'Chennai');

Display the list of customers whose age is greater than 50 and does not reside in Bangalore.

SELECT * FROM customers
WHERE age > 50 AND caddress <> 'Bangalore';

You want to retrieve data for all the orders who made order '1-12-90' having prices 4000 - 6000 and sort the column in descending order on price.

SELECT * FROM orders
WHERE orderdate = '1990-12-01'
AND price BETWEEN 4000 AND 6000
ORDER BY price DESC;

Display order details in format Cusid | Price (sum of price) | Count (count of qty)

SELECT custid AS Cusid,
SUM(price) AS [Price (sum of price)],
SUM(qty) AS [Count (count of qty)]
FROM orders GROUP BY custid
ORDER BY custid;

Display above details only for price > 4000

SELECT
    custid AS Cusid,
    SUM(price) AS [Price (sum of price)],
    SUM(qty) AS [Count (count of qty)]
FROM orders GROUP BY custid
HAVING SUM(price) > 4000
ORDER BY custid;
