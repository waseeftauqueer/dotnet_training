-- queries
-- 1
select distinct c.custid,c.custname,c.email,c.city
from customers c
join orders o
on c.custid = o.custid
where o.orderdate >=getdate()-30

-- 2
select top 3 p.productid,p.productname,sum(od.qty*p.price) as totsales
from products p
join orderdetails od on p.productid = od.productid
join orders o on od.orderid =o.orderid
group by p.productid, p.productname
order by totsales desc

-- 3
select c.city,count(distinct c.custid) as custcount,count(o.orderid) as ordcount
from customers c
left join orders o
on c.custid = o.custid
group by c.city

-- 4
select od.orderid
from orderdetails od
group by od.orderid
having count(od.productid) > 2

-- 5
select od.orderid,sum(od.qty*p.price) as totamnt
from orderdetails od
join products p
on od.productid=p.productid
group by od.orderid
having sum(od.qty*p.price)>10000

-- 6
select c.custid, c.custname from customers c
join orders o
on c.custid=o.custid
join orderdetails od
on o.orderid=od.orderid
group by c.custid,c.custname, od.productid
having count(*)> 1

-- 7
select o.custid as empid,count(o.orderid) as totorders,
min(o.orderdate) as firstdate,max(o.orderdate) as lastdate
from orders o group by o.custid

-- 2
alter function fn_getcustomerlevel(@custid int)
returns varchar(20)
as begin
declare @total decimal(10,2);
declare @level varchar(40);
 
select @total = sum(od.qty*p.price)
from orders o
join orderdetails od on o.orderid = od.orderid
join products p on od.productid = p.productid
where o.custid = @custid;
 
if @total > 100000
set @level ='platinum';
else if @total >= 50000 and @total <= 100000
set @level ='gold';
else
set @level ='silver';
return @level
end;
go

alter procedure sp_updateproductprice
@productid int,
@newprice decimal(10,2)
as
begin
set nocount on;
if (@newprice <= 0)
begin
throw 50001, 'new price must be greater than 0.',1;
end;
if not exists(select 1 from products where productid=@productid)
begin
throw 50002, 'product does not exist.', 1;
end;
declare @oldprice decimal(10,2);
select @oldprice =price from products where productid = @productid;
insert into pricehistory (productid, oldprice) values (@productid, @oldprice);
update products set price =@newprice where productid = @productid;
print 'price updated successfully.';
end;

-- 2
alter procedure sp_searchorders
@custname varchar(100) = null, @city varchar(100) = null,
@productname varchar(100) = null, @startdate date = null, @enddate date = null
as
begin
select o.orderid, o.orderdate,o.status,c.custname, c.city,p.productname, od.qty,(od.qty *p.price) as itemtotal
from orders o
join customers c on o.custid = c.custid
join orderdetails od on o.orderid = od.orderid
join products p on od.productid = p.productid
where (@custname is null or c.custname like '%' + @custname + '%')
and (@city is null or c.city = @city)
and (@productname is null or p.productname like '%' + @productname + '%')
and (@startdate is null or o.orderdate >= @startdate)
and (@enddate is null or o.orderdate <= @enddate);
end;
